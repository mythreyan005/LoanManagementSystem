CREATE TABLE LOAN_TB (
    LOANID VARCHAR2(12 BYTE) PRIMARY KEY,
    APPLICANTNAME VARCHAR2(50 BYTE) NOT NULL,
    LOANTYPE VARCHAR2(30 BYTE) NOT NULL,
    LOANAMOUNT NUMBER NOT NULL,
    APPLICATIONDATE DATE NOT NULL,
    STATUS VARCHAR2(20 BYTE) NOT NULL,
    REMARKS VARCHAR2(100 BYTE)
);
CREATE SEQUENCE LOAN_SEQ
START WITH 10
MAXVALUE 99
INCREMENT BY 1
NOCACHE
NOCYCLE;
